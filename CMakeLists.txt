cmake_minimum_required(VERSION 3.24)
project(llm_inference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LLM_CUDA "Build NVIDIA CUDA runtime" ON)
option(LLM_SANITIZE "Enable host address/undefined sanitizers" OFF)
include(CheckLanguage)
if(LLM_CUDA)
 check_language(CUDA)
 if(NOT CMAKE_CUDA_COMPILER)
  message(FATAL_ERROR "CUDA compiler not found. Use -DLLM_CUDA=OFF for explicit host-only tests.")
 endif()
 if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 75 80 86 89)
 endif()
 enable_language(CUDA)
 find_package(CUDAToolkit REQUIRED)
 set(CMAKE_CUDA_STANDARD 17)
 set(CMAKE_CUDA_STANDARD_REQUIRED ON)
endif()
add_library(llm_core src/tensor.cpp src/kv_cache.cpp src/scheduler.cpp src/quantization.cpp src/weight_loader.cpp)
target_include_directories(llm_core PUBLIC include)
target_compile_options(llm_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
if(LLM_SANITIZE)
 target_compile_options(llm_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(llm_core PUBLIC -fsanitize=address,undefined)
endif()
if(LLM_CUDA)
 target_compile_definitions(llm_core PUBLIC LLM_CUDA=1)
 target_link_libraries(llm_core PUBLIC CUDA::cudart)
endif()
enable_testing()
add_executable(test_systems tests/test_systems.cpp)
target_link_libraries(test_systems PRIVATE llm_core)
add_test(NAME systems COMMAND test_systems)
set_tests_properties(systems PROPERTIES TIMEOUT 30)
if(LLM_CUDA)
 add_library(llm_kernels kernels/gemm.cu kernels/ops.cu)
 target_include_directories(llm_kernels PUBLIC include)
 target_link_libraries(llm_kernels PUBLIC llm_core CUDA::cublas)
 set_target_properties(llm_kernels PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
endif()
option(LLM_TOKENIZER "Build native SentencePiece tokenizer" ${LLM_CUDA})
if(LLM_TOKENIZER)
 set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
 include(FetchContent)
 set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
 set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(sentencepiece GIT_REPOSITORY https://github.com/google/sentencepiece.git GIT_TAG v0.2.0 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(sentencepiece)
 add_library(llm_tokenizer src/tokenizer.cpp)
 target_include_directories(llm_tokenizer PUBLIC include ${sentencepiece_SOURCE_DIR}/src)
 target_link_libraries(llm_tokenizer PUBLIC sentencepiece-static)
 add_executable(tokenizer_dump tests/tokenizer_dump.cpp)
 target_link_libraries(tokenizer_dump PRIVATE llm_tokenizer)
endif()
add_executable(test_weights tests/test_weights.cpp)
target_link_libraries(test_weights PRIVATE llm_core)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME weight_format COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_conversion.py $<TARGET_FILE:test_weights>)
endif()
if(LLM_CUDA)
 add_library(llm_model src/model.cpp)
 target_link_libraries(llm_model PUBLIC llm_kernels)
endif()
